    sensor/.style={
        rectangle, rounded corners=2.5pt,

]
    % Sensor Layer
    \node (cam) [sensor] at (-1.78, 0) {\textbf{\scriptsize Rear Camera (CameraX)}\\[-1.5pt]\tiny $N=1$ Bounded Backpressure};
    \node (imu) [sensor] at (1.78, 0) {\textbf{\scriptsize Built-in IMU Sensors}\\[-1.5pt]\tiny Accel + Gyro (\texttt{SENSOR\_DELAY\_GAME})};

    % Processing Pipeline Stages
    \node (s1) [stage, below=0.25cm of $(cam.south)!0.5!(imu.south)$] {\textbf{\scriptsize 1. SPATIAL PERCEPTION \& EXTRACTION}\\[-1.5pt]\tiny $32 \times 24$ Luminance Grid, Sobel Energy ($E > 38f$), 4-Connectivity BFS};
    
    \node (s2) [stage, below=0.22cm of s1] {\textbf{\scriptsize 2. GEOMETRIC GROUND VALIDATION}\\[-1.5pt]\tiny Pinhole Ground Intersection ($Y_w=0$), Pitch/Roll Correction, $R(Z) = 0.04 + 0.015Z$};
    
    \node (s3) [stage, below=0.22cm of s2] {\textbf{\scriptsize 3. MULTI-TARGET TRACKING \& MOTION FUSION}\\[-1.5pt]\tiny Mahalanobis Gating, Constant-Velocity Kalman Filter, Step Detector \& State Machine};
    
    \node (s4) [stage, below=0.22cm of s3] {\textbf{\scriptsize 4. COLLISION ASSESSMENT \& DECISION ARBITRATION}\\[-1.5pt]\tiny Analytical CPA / Quadratic TTC, Spatial Risk ($0\text{--}100$), Corridor Clearance Hysteresis};
    
    \node (s5) [stage, below=0.22cm of s4] {\textbf{\scriptsize 5. HAZARD LIFECYCLE \& WARNING GENERATION}\\[-1.5pt]\tiny 9-State Hazard FSM, 6 Severity Tiers, 7 Spatial Sectors, 1.5s Cooldown Gate};

    % Multimodal Output
    \node (sp) [pres, below=0.25cm of s5.south west, xshift=1.06cm] {\textbf{\scriptsize Speech (TTS)}\\[-1.5pt]\tiny Audio Ducking};
    \node (aud) [pres, right=0.22cm of sp] {\textbf{\scriptsize Spatial Audio}\\[-1.5pt]\tiny 44.1\,kHz Stereo PCM};
    \node (hap) [pres, right=0.22cm of aud] {\textbf{\scriptsize Tactile Haptic}\\[-1.5pt]\tiny VibrationEffect};

    % Telemetry Sink
    \node (telem) [stage, below=0.22cm of aud, fill=gray!8] {\textbf{\scriptsize 6. ANDROID TELEMETRY \& EVALUATION SINK}\\[-1.5pt]\tiny Per-Frame JSONL Logging (20 Fields), Embedded HTTP/SSE Stream (Port 8080)};
